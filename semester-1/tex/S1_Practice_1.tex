\PracticeHeader{1}{1}{Множество}

\Task{Задание 1}
Задайте множество формулой.
\begin{numlist}
    \item $\{2,4,8,16,32,64,\ldots\}$
    \item $\{\ldots,-6,-3,0,3,6,9,12,15,\ldots\}$
    \item $\{0,1,4,9,16,25,36,\ldots\}$
    \item $\{3,4,5,6,7,8\}$
    \item $\{1,3,9,27,81,\ldots\}$
    \item $\{2,5,8,11,14,17,\ldots\}$
    \item $\{1,3,6,10,15,21,\ldots\}$
\end{numlist}

\Task{Задание 2}
Даны множества $A,B$. Найдите для них $\cup$, $\cap$, $\oplus$, $\setminus$.
\begin{numlist}
    \item $A=\{1,2,3\},\quad B=\{1,3,5,7\}$
    \item $A=\{x \mid x\in\mathbb{R}\},\quad B=\{x \mid x\in\mathbb{N}\}$
    \item $A=\{0,1,2,3,4\},\quad B=\{3,4,5,6\}$
    \item $A=\{x\in\mathbb{Z}\mid x \text{ чётный}\},\quad B=\{x\in\mathbb{Z}\mid 3\mid x\}$
    \item $A=[-2,5],\quad B=(3,\infty)$
\end{numlist}

\Task{Задание 3}
Дано множество $A=\{a,\{a\},b,\varnothing\}$. Какие из следующих утверждений верны:
\begin{numlist}
    \item $a \in A$
    \item $a \subset A$
    \item $b \in A$
    \item $b \subset A$
    \item $\varnothing \in A$
    \item $\varnothing \subset A$
    \item $\{a\}\in A$
    \item $\{a\}\subset A$
    \item $\{b,\varnothing\}\subset A$
    \item $\{a,b\}\in A$
\end{numlist}

\clearpage

\Task{Задание 4}
Дан универсум $U=\{1,2,3,4,5,6,7,8,9\}$ и его подмножества:
\begin{itemize}
    \item $A=\{1,2,3\}$
    \item $B=\{x\in U \mid 2\mid x\}$
    \item $C=\{x\in U \mid x \text{ – простое число}\}$
    \item $D=A\oplus B$
\end{itemize}
Найдите:
\begin{numlist}
    \item $(A \cup B)\cap D$
    \item $B \cap C$
    \item $A^{c}\cap U$ \quad(то же самое, что $U\setminus A$)
    \item $(C \setminus B)\ \oplus\ D$
\end{numlist}

\Task{Задание 5*}
\begin{numlist}
    \item Докажите, что для любых множеств $A,B$ верно тождество симметрической разности:
    \[
       A\oplus B=(A\cup B)\setminus(A\cap B).
    \]
    \item Пусть $U=\{1,2,\dots,60\}$. Обозначим
    \[
      A=\{x\in U \mid 2\mid x\},\quad
      B=\{x\in U \mid 3\mid x\},\quad
      C=\{x\in U \mid 5\mid x\}.
    \]
    Найдите $|A\cup B\cup C|$ и $|U\setminus(A\cup B\cup C)|$.
    \item Докажите дистрибутивность пересечения относительно симметрической разности:
    \[
      A\cap(B\oplus C)=(A\cap B)\oplus(A\cap C),
    \]
    а также укажите (с доказательством), являются ли операции $\cup$, $\cap$, $\oplus$ коммутативными и ассоциативными.
\end{numlist}
